ALTER procedure [dbo].[hizala]
as 
begin

select verilen_urun.musteri_kodu,sum(urun_tutar)as uruntutar,sum(verilen_para)as verilenpara
from para
inner join verilen_urun 
on verilen_urun.musteri_kodu=para.musteri_kodu

group by verilen_urun.musteri_kodu
order by sum(verilen_para) desc
end

---------------------------------------------------------------------------------------------
create procedure hizala2
(@musteri_kodu int)
as 

begin

select verilen_urun.musteri_kodu,sum(urun_tutar)as uruntutar,sum(verilen_para)as verilenpara
from para
inner join verilen_urun 
on verilen_urun.musteri_kodu=para.musteri_kodu
where verilen_urun.musteri_kodu=@musteri_kodu
group by verilen_urun.musteri_kodu
order by sum(verilen_para) desc
end
